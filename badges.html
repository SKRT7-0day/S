async function getBadges() { const userId = document.getElementById('userIdInput').value.trim(); const container = document.getElementById('badgesResult'); if (!userId) return; container.innerHTML = 'جاري التحميل...'; const res = await fetch(`/api/badges/${userId}`); const data = await res.json(); if (!data.success) { container.innerHTML = data.message; return; } if (data.badges.length === 0) { container.innerHTML = 'لا توجد شارات لهذا الحساب.'; return; } // عرض صور الشارات container.innerHTML = data.badges.map(badge => `
${badge.description}
`).join(''); }
